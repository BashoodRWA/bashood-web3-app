### Paso 2: Deploy PaymentSplitter Contract

```solidity
// OPCIÓN A: OpenZeppelin PaymentSplitter
import "@openzeppelin/contracts/finance/PaymentSplitter.sol";

address[] memory wallets = new address[](4);
wallets[0] = developmentWallet;  // 45%
wallets[1] = operationsWallet;   // 25%
wallets[2] = marketingWallet;    // 20%
wallets[3] = treasuryWallet;     // 10%

uint256[] memory shares = new uint256[](4);
shares[0] = 45;
shares[1] = 25;
shares[2] = 20;
shares[3] = 10;

PaymentSplitter splitter = new PaymentSplitter(wallets, shares);

// Usar splitter como projectWallet en presale
presale = new BashoodPresaleFinal(
    bhtToken,
    nftContract,
    referralContract,
    address(splitter),  // ← fondos se distribuyen automáticamente
    ethers.parseEther("0.1"),
    ethers.parseUnits("100", 18),
    startTime,
    endTime,
    1000
);
```
