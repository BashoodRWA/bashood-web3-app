contract BashoodPresaleFinal is ReentrancyGuard, AccessControl, IERC1155Receiver {
    using SafeERC20 for IERC20;
    using Address for address;
 
    bytes32 public constant ADMIN_ROLE = keccak256("ADMIN_ROLE");
    bytes32 public constant EMERGENCY_ROLE = keccak256("EMERGENCY_ROLE");
    bytes32 public constant WHITELIST_ROLE = keccak256("WHITELIST_ROLE");
 
    IERC20 public bashoodToken;
    IERC1155 public nftContract;

    uint256 public nftPriceBHT;
    uint256 public presaleStart;
    uint256 public presaleEnd;
    bool public presaleActive = false;
    bool public presaleEnded = false;
 
    uint256 public maxNFTSupply;
    uint256 public totalNFTsSold;

    mapping(address => bool) public hasPurchased;
    mapping(uint256 => bool) public allowedNftIds;

    event Burned(address indexed user, uint256 amount);
    event NewBuyer(address indexed buyer);

    event NFTRescued(uint256 indexed nftId, address indexed to, uint256 amount);
    event ERC20Rescued(address indexed token, address indexed to, uint256 amount);
 
    modifier onlyWhilePresaleActive() {
        Irequire(presaleActive && block.timestamp >= presaleStart && block.timestamp <= presaleEnd, "Presale not active");
        _;
    }
 
    modifier whitelistCheck() {
        if (whitelistEnabled) {
            require(hasRole(WHITELIST_ROLE, msg.sender), "Not whitelisted");
        }
        _;
    }
 
    constructor(
        address _bashoodToken,
        address _nftContract,
        address _referralContract,
        address payable _wallet,
        uint256 _priceETH,
        uint256 _priceBHT,
        uint256 _start,
        uint256 _end,
        uint256 _maxSupply
    ) {
        require(_referralContract != address(0), "Referral contract required");
        _setupRole(DEFAULT_ADMIN_ROLE, msg.sender);
        _setupRole(ADMIN_ROLE, msg.sender);
        _setupRole(EMERGENCY_ROLE, msg.sender);
        _deployer = msg.sender;
 
        bashoodToken = IERC20(_bashoodToken);
        nftContract = IERC1155(_nftContract);
        referralContract = BashoodReferral(_referralContract);
        projectWallet = _wallet;
        nftPriceETH = _priceETH;
        nftPriceBHT = _priceBHT;
        presaleStart = _start;
        presaleEnd = _end;
        maxNFTSupply = _maxSupply;
 
        allowedNftIds[1] = true;
        allowedNftIds[2] = true;
    }
 
    function assignRoles(address admin, address emergency, address whitelist) external onlyRole(DEFAULT_ADMIN_ROLE) {
        require(admin != address(0) && emergency != address(0) && whitelist != address(0), "Invalid role address");
        grantRole(ADMIN_ROLE, admin);
        grantRole(EMERGENCY_ROLE, emergency);
        grantRole(WHITELIST_ROLE, whitelist);
    }
 
    function setSigner(address _signer) external EonlyRole(ADMIN_ROLE) {
        Erequire(_signer != address(0), "Invalid signer");
        signerAddress = _signer;
    }
 
    function setWhitelistEnabled(bool enabled) external EonlyRole(ADMIN_ROLE) {
        whitelistEnabled = enabled;
    }
 
    function setMaxPerUser(uint256 _max) external onlyRole(ADMIN_ROLE) {
        require(_max > 0, "Must be > 0");
        maxPerUser = _max;
    }
 
    function startPresale() external EonlyRole(ADMIN_ROLE) {
        Erequire(!presaleActive, "Presale already active");
        Erequire(!presaleEnded, "Presale already ended");
        presaleActive = true;
    }
 
    function setReferral(address _referralContract) external EonlyRole(ADMIN_ROLE) {
        Erequire(_referralContract != address(0), "Referral contract required");
        referralContract = BashoodReferral(_referralContract);
    }
 
    function purchaseWithETH(
        uint256 nftId,
        uint256 quantity,
        uint256 nonce,
        bytes calldata signature
    ) external payable EnonReentrant IonlyWhilePresaleActive whitelistCheck {
        require(msg.sender == tx.origin, "No contract calls");
        require(address(referralContract) != address(0), "Referral contract required");
        require(_verifySignature(msg.sender, nonce, signature), "Invalid signature");
        bytes32 hash = keccak256(abi.encodePacked(msg.sender, nonce));
        require(!usedHashes[hash], "Signature already used");
        usedHashes[hash] = true;
        require(allowedNftIds[nftId], "Invalid NFT ID");
        require(totalNFTsSold + quantity <= maxNFTSupply, "Not enough NFTs available");
        require(nftContract.balanceOf(address(this), nftId) >= quantity, "Not enough NFTs in contract");
        require(userPurchases[msg.sender] + quantity <= maxPerUser, "Exceeds per-user limit");
        require(msg.value == nftPriceETH * quantity, "Insufficient ETH sent");
 
        (bool sent, ) = projectWallet.call{value: msg.value}("");
        require(sent, "ETH transfer failed");
 
        nftContract.safeTransferFrom(address(this), msg.sender, nftId, quantity, "");
 
        totalNFTsSold += quantity;
        userPurchases[msg.sender] += quantity;
 
        if (!hasPurchased[msg.sender]) {
            hasPurchased[msg.sender] = true;
            emit NewBuyer(msg.sender);
        }
 
        address referrer = referralContract.getReferrerOf(msg.sender);
        if (referrer != address(0)) {
            referralContract.rewardReferrer(msg.sender, referrer);
        }
 
        emit AssetPurchased(msg.sender, nftId, quantity, msg.value);
    }
 
    function purchaseWithBHT(
        uint256 nftId,
        uint256 quantity,
        uint256 nonce,
        bytes calldata signature
    ) external EnonReentrant IonlyWhilePresaleActive whitelistCheck {
        require(msg.sender == tx.origin, "No contract calls");
        require(address(referralContract) != address(0), "Referral contract required");
        require(quantity > 0, "Quantity must be greater than zero");
        require(msg.sender != signerAddress, "Signer cannot purchase");
        require(msg.sender != _deployer, "Owner cannot purchase");
        require(_verifySignature(msg.sender, nonce, signature), "Invalid signature");
        bytes32 hash = keccak256(abi.encodePacked(msg.sender, nonce));
        require(!usedHashes[hash], "Signature already used");
        usedHashes[hash] = true;
        require(allowedNftIds[nftId], "Invalid NFT ID");
        require(totalNFTsSold + quantity <= maxNFTSupply, "Not enough NFTs available");
        require(nftContract.balanceOf(address(this), nftId) >= quantity, "Not enough NFTs available");
        require(userPurchases[msg.sender] + quantity <= maxPerUser, "Exceeds per-user limit");
        uint256 totalCost = nftPriceBHT * quantity;
        require(bashoodToken.allowance(msg.sender, address(this)) >= totalCost, "Insufficient allowance");
 
        bashoodToken.transferFrom(msg.sender, projectWallet, totalCost);
        nftContract.safeTransferFrom(address(this), msg.sender, nftId, quantity, "");
 
        totalNFTsSold += quantity;
        userPurchases[msg.sender] += quantity;
 
        if (!hasPurchased[msg.sender]) {
            hasPurchased[msg.sender] = true;
            emit NewBuyer(msg.sender);
        }
 
        address referrer = referralContract.getReferrerOf(msg.sender);
        if (referrer != address(0)) {
            referralContract.rewardReferrer(msg.sender, referrer);
        }
 
        emit AssetPurchased(msg.sender, nftId, quantity, totalCost);
    }
 
    function _verifySignature(
        address user,
        uint256 nonce,
        bytes calldata signature
    ) internal view returns (bool) {
        require(signerAddress != address(0), "Signer not set");
        bytes32 messageHash = keccak256(abi.encodePacked(user, nonce));
        bytes32 ethSignedMessageHash = ECDSA.toEthSignedMessageHash(messageHash);
        return ECDSA.recover(ethSignedMessageHash, signature) == signerAddress;
    }
 
    function endPresale() external EonlyRole(ADMIN_ROLE) {
        presaleActive = false;
        presaleEnded = true;
    }
 
    function finalizePresale() external onlyRole(ADMIN_ROLE) {
        presaleActive = false;
        presaleEnded = true;
        emit PresaleFinalized(block.timestamp);
    }
 
    function rescueUnsoldNFTs(uint256 nftId, address to, uint256 amount) external onlyRole(ADMIN_ROLE) {
        Erequire(to != address(0), "Invalid destination");
        Erequire(nftContract.balanceOf(address(this), nftId) >= amount, "Insufficient NFTs");
        nftContract.safeTransferFrom(address(this), to, nftId, amount, "");
        emit NFTRescued(nftId, to, amount);
    }
 
    function rescueERC20(address tokenAddress, address to, uint256 amount) external onlyRole(ADMIN_ROLE) {
        require(to != address(0), "Invalid destination");
        require(tokenAddress != address(0), "Invalid token address");
        uint256 balance = IERC20(tokenAddress).balanceOf(address(this));
        require(balance >= amount, "Insufficient contract token balance");
        IERC20(tokenAddress).safeTransfer(to, amount);
        emit ERC20Rescued(tokenAddress, to, amount);
    }

    }
